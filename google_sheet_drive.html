พร้อมอัปโหลดเข้า Google Drive Folder

โค้ด Google Apps Script (เชื่อมต่อ Sheet + Drive Folder)

วิธีนำไปใช้งานจริง 100%:

  1. เปิด Google Sheet สร้างแท็บชื่อ Orders
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดด้านล่างนี้ลงไป แล้วเปลี่ยน SLIP_FOLDER_ID เป็น ID โฟลเดอร์ Google Drive ของคุณ
  4. กด ทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment) > เว็บแอป (Web app) เลือกสิทธิ์ผู้เข้าถึงเป็น ทุกคน (Anyone)
  5. นำ Web App URL ที่ได้มาวางในช่องตั้งค่าบนหน้าเว็บนี้
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Orders") || ss.insertSheet("Orders");
    
    // สร้างหัวตารางอัตโนมัติถ้ายังไม่มี
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Order ID", "วันที่-เวลา", "ชื่อลูกค้า", "เบอร์โทร", "ที่อยู่", "รายการ SKU", "ยอดรวม", "วิธีชำระเงิน", "ลิงก์รูปสลิป (Google Drive)"]);
    }

    var slipUrl = "ไม่มีการแนบสลิป (COD)";
    
    // บันทึกรูปสลิปลง Google Drive โฟลเดอร์แยกต่างหาก (ถ้ามีไฟล์แนบ Base64)
    if (data.slipBase64 && data.slipFileName) {
      var folderId = data.slipFolderId || "ใส่_FOLDER_ID_ของคุณที่นี่";
      var folder = DriveApp.getFolderById(folderId);
      var contentType = data.slipBase64.substring(5, data.slipBase64.indexOf(';'));
      var bytes = Utilities.base64Decode(data.slipBase64.substr(data.slipBase64.indexOf('base64,') + 7));
      var blob = Utilities.newBlob(bytes, contentType, data.orderId + "_" + data.slipFileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      slipUrl = file.getUrl();
    }

    // บันทึกแถวข้อมูลคำสั่งซื้อลง Google Sheet
    sheet.appendRow([
      data.orderId,
      data.timestamp,
      data.customerName,
      data.phone,
      data.address,
      data.itemsSummary,
      data.totalAmount,
      data.paymentMethod,
      slipUrl
    ]);

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      orderId: data.orderId,
      slipUrl: slipUrl
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
